```js
// Regenerates src/lib/assistant/kb-data.ts from MDX. Runs on every build via `prebuild`.
import { readdirSync, readFileSync, writeFileSync, mkdirSync } from "node:fs";
import path from "node:path";

const ROOT = process.cwd();
const SECTIONS = { projects: "projects", research: "research", writing: "writing" };

function cleanBody(raw) {
  return raw
    .split("\n")
    .filter((line) => !line.startsWith("import ") && line.trim() !== "")
    .join("\n")
    .replace(/export const metadata = \{[\s\S]*?\};/, "")
    .trim();
}

function splitSections(body) {
  const parts = body.split(/^## /m);
  const head = parts.shift().trim();
  const out = [];
  if (head) out.push({ heading: "", text: head });
  for (const part of parts) {
    const nl = part.indexOf("\n");
    out.push({ heading: part.slice(0, nl).trim(), text: part.slice(nl + 1).trim() });
  }
  return out.filter((s) => s.text.length > 0);
}

const chunks = [];
for (const [dir, routeBase] of Object.entries(SECTIONS)) {
  const dirPath = path.join(ROOT, "src", "content", dir);
  for (const file of readdirSync(dirPath).filter((f) => f.endsWith(".mdx"))) {
    const raw = readFileSync(path.join(dirPath, file), "utf8");
    const metaMatch = raw.match(/title:\s*"([^"]+?)"/);
    const slug = file.replace(/\.mdx$/, "");
    const title = metaMatch ? metaMatch[1] : slug;
    for (const s of splitSections(cleanBody(raw))) {
      chunks.push({ sourceRoute: `/${routeBase}/${slug}`, sourceTitle: title, heading: s.heading, text: s.text.slice(0, 1200) });
    }
  }
}

mkdirSync(path.join(ROOT, "src", "lib", "assistant"), { recursive: true });
writeFileSync(
  path.join(ROOT, "src", "lib", "assistant", "kb-data.ts"),
  `// GENERATED by scripts/build-kb.mjs — do not edit by hand.\nimport type { KbChunk } from "./kb";\n\nexport const KB_CHUNKS: KbChunk[] = ${JSON.stringify(chunks, null, 2)};\n`,
);
console.log(`wrote ${chunks.length} chunks`);
```

```ts
// src/lib/assistant/kb.ts
export interface KbChunk {
  sourceRoute: string;
  sourceTitle: string;
  heading: string;
  text: string;
}

const STOPWORDS = new Set(
  "a,an,the,and,or,but,of,to,in,on,for,with,what,which,who,how,does,do,is,are,he,his,him,has,have,had,was,were,be,been,by,about,can,could,should,would,there,their,they,it,its,as,at,from,that,this,these,those,you,your,i,me,my,we,our".split(","),
);

export function tokenize(s: string): string[] {
  return s
    .toLowerCase()
    .replace(/[^a-z0-9\s]/g, " ")
    .split(/\s+/)
    .filter((w) => w.length > 2 && !STOPWORDS.has(w));
}

export function scoreChunk(queryTokens: string[], chunk: KbChunk): number {
  const hay = new Set(tokenize(`${chunk.sourceTitle} ${chunk.heading} ${chunk.text}`));
  let score = 0;
  for (const t of queryTokens) if (hay.has(t)) score += 1;
  return score;
}

export interface Retrieval {
  chunks: KbChunk[];
  topScore: number;
}

export function retrieve(query: string, all: KbChunk[], maxChunks: number): Retrieval {
  const tokens = tokenize(query);
  const ranked = all
    .map((chunk) => ({ chunk, score: scoreChunk(tokens, chunk) }))
    .filter((r) => r.score > 0)
    .sort((a, b) => b.score - a.score)
    .slice(0, maxChunks);
  return { chunks: ranked.map((r) => r.chunk), topScore: ranked[0]?.score ?? 0 };
}

export function buildContext(chunks: KbChunk[], wordBudget: number): string {
  let words = 0;
  const out: string[] = [];
  for (const c of chunks) {
    const body = `[${c.sourceTitle}](${c.sourceRoute})${c.heading ? ` — ${c.heading}` : ""}\n${c.text}`;
    const n = body.split(/\s+/).length;
    if (words + n > wordBudget) break;
    words += n;
    out.push(body);
  }
  return out.join("\n\n");
}
```

```ts
// src/lib/assistant/ratelimit.ts
export interface RateLimitResult {
  ok: boolean;
  retryAfterMs: number;
}

export function createRateLimiter(opts: {
  limit: number;
  windowMs: number;
  now?: () => number;
}) {
  const hits = new Map<string, number[]>();
  const now = opts.now ?? Date.now;
  return {
    check(ip: string): RateLimitResult {
      const t = now();
      const arr = (hits.get(ip) ?? []).filter((ts) => t - ts < opts.windowMs);
      if (arr.length >= opts.limit) {
        const oldest = arr[0];
        return { ok: false, retryAfterMs: opts.windowMs - (t - oldest) };
      }
      arr.push(t);
      hits.set(ip, arr);
      return { ok: true, retryAfterMs: 0 };
    },
  };
}
```

```ts
import { NextRequest } from "next/server";
import { assistantConfig, openRouterKey, openRouterModel } from "@/lib/assistant/config";
import { KB_CHUNKS } from "@/lib/assistant/kb-data";
import { buildContext, retrieve } from "@/lib/assistant/kb";
import { buildMessages, type ChatTurn } from "@/lib/assistant/prompt";
import { createRateLimiter } from "@/lib/assistant/ratelimit";
import { siteUrl } from "@/lib/site";

const limiter = createRateLimiter({
  limit: assistantConfig.rateLimit,
  windowMs: assistantConfig.rateWindowMs,
});

const FALLBACK_SUGGESTIONS = [
  { q: "What agentic systems has he built?", route: "/projects/aura" },
  { q: "Explain the architecture of AURA.", route: "/projects/aura" },
  { q: "What has he published?", route: "/research/insulator-fault-detection" },
];

function sse(data: string): string {
  return `data: ${data}\n\n`;
}

function clientIp(req: NextRequest): string {
  return (
    req.headers.get("x-forwarded-for")?.split(",")[0]?.trim() ??
    req.headers.get("x-real-ip") ??
    "unknown"
  );
}

export async function POST(req: NextRequest) {
  let body: { message?: unknown; history?: unknown };
  try {
    body = await req.json();
  } catch {
    return Response.json({ error: "Invalid JSON body." }, { status: 400 });
  }
  const message = typeof body.message === "string" ? body.message.trim() : "";
  const history = Array.isArray(body.history)
    ? (body.history as ChatTurn[]).filter(
        (t) => (t.role === "user" || t.role === "assistant") && typeof t.content === "string",
      )
    : [];
  if (!message || message.length > assistantConfig.maxInputChars) {
    return Response.json(
      { error: `Message must be 1–${assistantConfig.maxInputChars} characters.` },
      { status: 400 },
    );
  }
  if (history.length > assistantConfig.maxTurns * 2) {
    return Response.json({ error: "Conversation too long — please start over." }, { status: 400 });
  }

  const rl = limiter.check(clientIp(req));
  if (!rl.ok) {
    return Response.json(
      { error: "Too many questions too quickly — please wait a minute and try again." },
      { status: 429 },
    );
  }

  const { chunks, topScore } = retrieve(message, KB_CHUNKS, assistantConfig.maxChunks);
  if (topScore < assistantConfig.minScore || chunks.length === 0) {
    return Response.json({
      fallback: true,
      message:
        "I don't have verified information on that in the public portfolio. Try one of these:",
      suggestions: FALLBACK_SUGGESTIONS,
    });
  }

  if (!openRouterKey) {
    return Response.json(
      {
        error:
          "The assistant is unavailable right now — browse the case studies or get in touch.",
      },
      { status: 503 },
    );
  }

  const context = buildContext(chunks, assistantConfig.wordBudget);
  const sources = chunks.map((c) => ({ title: c.sourceTitle, route: c.sourceRoute }));

  const controller = new AbortController();
  const timeout = setTimeout(() => controller.abort(), assistantConfig.requestTimeoutMs);
  let upstream: Response;
  try {
    upstream = await fetch("https://openrouter.ai/api/v1/chat/completions", {
      method: "POST",
      signal: controller.signal,
      headers: {
        Authorization: `Bearer ${openRouterKey}`,
        "Content-Type": "application/json",
        "HTTP-Referer": siteUrl,
        "X-Title": "Portfolio assistant",
      },
      body: JSON.stringify({
        model: openRouterModel,
        stream: true,
        max_tokens: assistantConfig.maxOutputTokens,
        messages: buildMessages([...history, { role: "user", content: message }], context),
      }),
    });
  } catch {
    clearTimeout(timeout);
    return Response.json({ error: "The assistant timed out — please try again." }, { status: 504 });
  }
  clearTimeout(timeout);

  if (!upstream.ok || !upstream.body) {
    return Response.json(
      { error: "The assistant is unavailable right now — browse the case studies or get in touch." },
      { status: 502 },
    );
  }

  const encoder = new TextEncoder();
  const stream = new ReadableStream({
    async start(ctrl) {
      const reader = upstream.body!.getReader();
      const decoder = new TextDecoder();
      let buf = "";
      try {
        for (;;) {
          const { done, value } = await reader.read();
          if (done) break;
          buf += decoder.decode(value, { stream: true });
          const lines = buf.split("\n");
          buf = lines.pop() ?? "";
          for (const line of lines) {
            const t = line.trim();
            if (!t.startsWith("data:")) continue;
            const payload = t.slice(5).trim();
            if (payload === "[DONE]") continue;
            try {
              const delta = JSON.parse(payload).choices?.[0]?.delta?.content;
              if (typeof delta === "string" && delta) {
                ctrl.enqueue(encoder.encode(sse(JSON.stringify({ token: delta }))));
              }
            } catch {
              // ignore malformed SSE lines
            }
          }
        }
        ctrl.enqueue(encoder.encode(sse(JSON.stringify({ sources }))));
        ctrl.enqueue(encoder.encode("data: [DONE]\n\n"));
      } finally {
        ctrl.close();
      }
    },
  });

  return new Response(stream, {
    headers: { "Content-Type": "text/event-stream", "Cache-Control": "no-cache" },
  });
}
```

```tsx
import Link from "next/link";

export interface Source {
  title: string;
  route: string;
}

export interface Message {
  id: number;
  role: "user" | "assistant";
  text: string;
  sources?: Source[];
}

export function ChatMessage({ message }: { message: Message }) {
  if (message.role === "user") {
    return (
      <p>
        <strong>You: </strong>
        {message.text}
      </p>
    );
  }
  return (
    <div>
      <p>{message.text}</p>
      {message.sources && message.sources.length > 0 && (
        <ul aria-label="Sources">
          {message.sources.map((s) => (
            <li key={s.route}>
              <Link href={s.route}>{s.title}</Link>
            </li>
          ))}
        </ul>
      )}
    </div>
  );
}
```

```tsx
"use client";

import { useRef, useState } from "react";
import { ChatMessage, type Message, type Source } from "./chat-message";

const SUGGESTIONS = [
  "What agentic systems has he built?",
  "Explain the architecture of AURA.",
  "What has he published?",
];

const DISCLAIMER = "Experimental assistant — answers are grounded in public portfolio content and may be incomplete.";

export function ChatPanel({ onClose }: { onClose: () => void }) {
  const [messages, setMessages] = useState<Message[]>([]);
  const [input, setInput] = useState("");
  const [busy, setBusy] = useState(false);
  const idRef = useRef(0);
  const nextId = () => ++idRef.current;

  async function send(text: string) {
    const clean = text.trim().slice(0, 1000);
    if (!clean || busy) return;
    const userMsg: Message = { id: nextId(), role: "user", text: clean };
    const history = [...messages, userMsg].map((m) => ({ role: m.role, content: m.text }));
    setMessages((prev) => [...prev, userMsg, { id: nextId(), role: "assistant", text: "" }]);
    setInput("");
    setBusy(true);
    try {
      const res = await fetch("/api/ask", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ message: clean, history: history.slice(0, -1) }),
      });
      const contentType = res.headers.get("content-type") ?? "";
      if (!res.ok || !contentType.includes("text/event-stream")) {
        const data = await res.json().catch(() => ({}));
        throw new Error(
          typeof data.message === "string"
            ? `${data.message} ${(data.suggestions ?? []).map((s: { q: string }) => s.q).join(" ")}`
            : typeof data.error === "string"
              ? data.error
              : "The assistant is unavailable right now.",
        );
      }
      const reader = res.body!.getReader();
      const decoder = new TextDecoder();
      let buf = "";
      for (;;) {
        const { done, value } = await reader.read();
        if (done) break;
        buf += decoder.decode(value, { stream: true });
        const lines = buf.split("\n\n");
        buf = lines.pop() ?? "";
        for (const line of lines) {
          const t = line.trim();
          if (!t.startsWith("data:")) continue;
          const payload = t.slice(5).trim();
          if (payload === "[DONE]") continue;
          try {
            const obj = JSON.parse(payload);
            if (typeof obj.token === "string") {
              const token = obj.token;
              setMessages((prev) => {
                const copy = [...prev];
                copy[copy.length - 1] = {
                  ...copy[copy.length - 1],
                  text: copy[copy.length - 1].text + token,
                };
                return copy;
              });
            } else if (Array.isArray(obj.sources)) {
              const sources = obj.sources as Source[];
              setMessages((prev) => {
                const copy = [...prev];
                copy[copy.length - 1] = { ...copy[copy.length - 1], sources };
                return copy;
              });
            }
          } catch {
            // ignore malformed events
          }
        }
      }
    } catch (err) {
      const text = err instanceof Error ? err.message : "The assistant is unavailable right now.";
      setMessages((prev) => {
        const copy = [...prev];
        copy[copy.length - 1] = { ...copy[copy.length - 1], text };
        return copy;
      });
    } finally {
      setBusy(false);
    }
  }

  return (
    <section aria-label="Ask about my work" aria-live="polite">
      <button type="button" onClick={onClose} aria-label="Close assistant">
        Close
      </button>
      {messages.length === 0 && (
        <ul>
          {SUGGESTIONS.map((q) => (
            <li key={q}>
              <button type="button" onClick={() => void send(q)} disabled={busy}>
                {q}
              </button>
            </li>
          ))}
        </ul>
      )}
      {messages.map((m) => (
        <ChatMessage key={m.id} message={m} />
      ))}
      <p>{DISCLAIMER}</p>
      <form
        onSubmit={(e) => {
          e.preventDefault();
          void send(input);
        }}
      >
        <label htmlFor="ask-input">Ask about my work</label>
        <input
          id="ask-input"
          value={input}
          onChange={(e) => setInput(e.target.value)}
          maxLength={1000}
          disabled={busy}
          autoComplete="off"
        />
        <button type="submit" disabled={busy || !input.trim()}>
          Send
        </button>
      </form>
    </section>
  );
}
```
